cmake_minimum_required(VERSION 3.20)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(fuurin_detector_sample LANGUAGES C)
target_sources(app PRIVATE src/main.c)
if(FUURIN_SAMPLE_MODEL)
  if(NOT EXISTS "${FUURIN_SAMPLE_MODEL}")
    message(FATAL_ERROR "FUURIN_SAMPLE_MODEL does not exist")
  endif()
  target_compile_definitions(app PRIVATE SAMPLE_HAS_MODEL=1)
  target_include_directories(app PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
  generate_inc_file_for_target(app ${FUURIN_SAMPLE_MODEL} ${CMAKE_CURRENT_BINARY_DIR}/generated/model.inc)
endif()
