cmake_minimum_required(VERSION 3.20)
if(NOT TARGET zephyr_interface)
  project(fuurin_audio_pipeline_host C)
  set(CMAKE_C_STANDARD 11)
  include(cmake/library.cmake)
  include(CTest)
  if(BUILD_TESTING AND ZAT_AEC_BACKEND STREQUAL "SPEEXDSP")
    add_subdirectory(tests/pipeline)
  endif()
  return()
endif()
if(CONFIG_FUURIN_AUDIO)
  zephyr_include_directories(include)
  if(NOT TARGET fvad)
    find_path(FUURIN_LIBFVAD_DIR NAMES src/fvad.c
      HINTS ${WEST_TOPDIR}/modules/lib/libfvad
      DOC "Root of the libfvad source checkout")
    if(NOT FUURIN_LIBFVAD_DIR)
      message(FATAL_ERROR "libfvad missing: west update libfvad, or set FUURIN_LIBFVAD_DIR")
    endif()
    add_subdirectory(${FUURIN_LIBFVAD_DIR}/src ${CMAKE_CURRENT_BINARY_DIR}/fvad)
    target_link_libraries(fvad PRIVATE zephyr_interface)
  endif()
  add_library(fuurin_audio_toolkit INTERFACE)
  target_include_directories(fuurin_audio_toolkit INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  if(CONFIG_FUURIN_AUDIO_AFE)
    include(cmake/library.cmake)
    zephyr_link_libraries(fuurin_audio_pipeline)
    target_link_libraries(fuurin_audio_toolkit INTERFACE fuurin_audio_pipeline)
  endif()
  if(CONFIG_FUURIN_AUDIO_DETECTOR)
  zephyr_library_named(fuurin_audio_detector)
  zephyr_include_directories(include)
  zephyr_library_sources(src/detector.cpp)
  target_link_libraries(fuurin_audio_toolkit INTERFACE fuurin_audio_detector)
  if(CONFIG_FUURIN_AUDIO_WAKEWORD)
    zephyr_library_sources(src/engine.cpp)
  else()
    zephyr_library_sources(src/engine_stub.cpp)
  endif()
  zephyr_library_link_libraries(fvad)
  if(CONFIG_FUURIN_AUDIO_WAKEWORD)
  set(FRONTEND ${ZEPHYR_TFLITE_MICRO_MODULE_DIR}/tensorflow/lite/experimental/microfrontend/lib)
  foreach(SOURCE fft.cc fft_util.cc kiss_fft_int16.cc filterbank.c filterbank_util.c
      frontend.c frontend_util.c log_lut.c log_scale.c log_scale_util.c
      noise_reduction.c noise_reduction_util.c pcan_gain_control.c
      pcan_gain_control_util.c window.c window_util.c)
    zephyr_library_sources(${FRONTEND}/${SOURCE})
  endforeach()
  endif()
  endif()
endif()
